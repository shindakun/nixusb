# Shared base packages, both machines. Official repos only.
# Install:  pacman -S --needed - < base.txt
# Lines starting with # and blank lines are ignored by the install scripts.
#
# The base install is repo-only on purpose: it runs as root from the live ISO,
# and makepkg will not run as root. The few things with no repo alternative are
# in aur.txt and are installed after first boot by install/aur.sh.
